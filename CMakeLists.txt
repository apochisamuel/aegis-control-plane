cmake_minimum_required(VERSION 3.20)
project(aegis_control_plane LANGUAGES CXX)
option(AEGIS_BUILD_TESTS "Build tests" ON)
option(AEGIS_BUILD_TOOLS "Build tools" ON)
option(AEGIS_BUILD_FUZZERS "Build libFuzzer targets" OFF)
option(AEGIS_ENABLE_SANITIZERS "Enable address and undefined sanitizers" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(GLOB_RECURSE AEGIS_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_library(aegis STATIC ${AEGIS_SOURCES})
target_include_directories(aegis PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(aegis PRIVATE -Wall -Wextra -Wpedantic)
if(AEGIS_ENABLE_SANITIZERS)
 target_compile_options(aegis PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(aegis PUBLIC -fsanitize=address,undefined)
endif()
if(AEGIS_BUILD_TOOLS)
 foreach(tool inspect_capture generate_examples policy_check)
  add_executable(${tool} tools/${tool}.cpp)
  target_link_libraries(${tool} PRIVATE aegis)
 endforeach()
endif()
if(AEGIS_BUILD_TESTS)
 enable_testing()
 foreach(test_name test_capture test_codec test_components)
  add_executable(${test_name} tests/${test_name}.cpp)
  target_link_libraries(${test_name} PRIVATE aegis)
  add_test(NAME ${test_name} COMMAND ${test_name})
 endforeach()
endif()
if(AEGIS_BUILD_FUZZERS)
 foreach(target capture_stream_fuzzer frame_sequence_fuzzer policy_blob_fuzzer extension_block_fuzzer)
  add_executable(${target} fuzz/${target}.cc)
  target_link_libraries(${target} PRIVATE aegis)
  target_compile_options(${target} PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
  target_link_options(${target} PRIVATE -fsanitize=fuzzer,address,undefined)
 endforeach()
endif()
